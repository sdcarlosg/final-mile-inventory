function doPost(e) {
  const lock = LockService.getScriptLock();
  lock.tryLock(10000);

  try {
    const jsonString = e.postData.contents;
    const items = JSON.parse(jsonString);
    const ss = SpreadsheetApp.getActiveSpreadsheet();

    // 1. Authenticate (Optional but recommended)
    // You can check items.passcode if you want strictly server-side checks

    // 2. Handle Sync Action
    if (items.action === "sync") {
      
      // --- Update Inventory Sheet ---
      const inventorySheet = ss.getSheetByName("Inventory") || ss.insertSheet("Inventory");
      // Clear old data (preserve headers if you want, usually row 1)
      if (inventorySheet.getLastRow() > 1) {
        inventorySheet.getRange(2, 1, inventorySheet.getLastRow() - 1, inventorySheet.getLastColumn()).clearContent();
      }
      // Ensure Headers exist
      if (inventorySheet.getLastRow() === 0) {
        inventorySheet.appendRow(["Asset ID", "Name", "Description", "Assigned To", "Status", "Date", "Date Assigned", "Date Returned", "Condition", "Incident Note", "Last Assigned To", "Last Date Assigned", "Last Date Returned"]);
      }

      const inventoryData = items.data.map(item => [
        item.id,
        item.nombre,
        item.descripcion,
        item.asignadoA,
        item.status,
        item.fecha,
        item.fechaAsignacion || "",
        item.fechaRetorno || "",
        item.condition || "",
        item.incidentNote || "",
        item.lastAssignedTo || "",
        item.lastDateAssigned || "",
        item.lastDateReturned || ""
      ]);

      if (inventoryData.length > 0) {
        inventorySheet.getRange(2, 1, inventoryData.length, 13).setValues(inventoryData);
      }

      // --- Update Employees Sheet ---
      const employeeSheet = ss.getSheetByName("Employees") || ss.insertSheet("Employees");
      // Clear old data
      if (employeeSheet.getLastRow() > 1) {
        employeeSheet.getRange(2, 1, employeeSheet.getLastRow() - 1, employeeSheet.getLastColumn()).clearContent();
      }
      // Ensure Headers exist (Added "Phone")
      if (employeeSheet.getLastRow() === 0) {
        employeeSheet.appendRow(["Badge ID", "Full Name", "Phone Number"]);
      }

      // Map data including the new Phone field
      const employeeData = items.employees.map(emp => [
        emp.id,
        emp.name,
        emp.phone || ""  // Handle newly added phone field
      ]);

      if (employeeData.length > 0) {
        employeeSheet.getRange(2, 1, employeeData.length, 3).setValues(employeeData);
      }

      // --- Update History Log Sheet ---
      const historySheet = ss.getSheetByName("History Log") || ss.insertSheet("History Log");
      if (historySheet.getLastRow() > 1) {
        historySheet.getRange(2, 1, historySheet.getLastRow() - 1, historySheet.getLastColumn()).clearContent();
      }
      if (historySheet.getLastRow() === 0) {
        historySheet.appendRow(["ID", "Timestamp", "Date", "Asset ID", "Asset Name", "Employee", "Action"]);
      }
      
      if (items.history && items.history.length > 0) {
        const historyData = items.history.map(h => [
          h.id, h.timestamp, h.dateStr, h.assetId, h.assetName, h.employeeName, h.action
        ]);
        historySheet.getRange(2, 1, historyData.length, 7).setValues(historyData);
      }

      return ContentService.createTextOutput(JSON.stringify({ status: "success" })).setMimeType(ContentService.MimeType.JSON);
    }

  } catch (e) {
    return ContentService.createTextOutput(JSON.stringify({ status: "error", error: e.toString() })).setMimeType(ContentService.MimeType.JSON);
  } finally {
    lock.releaseLock();
  }
}

function doGet(e) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();

  // Inventory
  const inventorySheet = ss.getSheetByName("Inventory");
  let inventory = [];
  if (inventorySheet && inventorySheet.getLastRow() > 1) {
    const data = inventorySheet.getRange(2, 1, inventorySheet.getLastRow() - 1, 13).getValues();
    inventory = data.map(row => ({
      id: row[0],
      nombre: row[1],
      descripcion: row[2],
      asignadoA: row[3],
      status: row[4],
      fecha: row[5],
      fechaAsignacion: row[6] || "",
      fechaRetorno: row[7] || "",
      condition: row[8] || "",
      incidentNote: row[9] || "",
      lastAssignedTo: row[10] || "",
      lastDateAssigned: row[11] || "",
      lastDateReturned: row[12] || ""
    }));
  }

  // Employees
  const employeeSheet = ss.getSheetByName("Employees");
  let employees = [];
  if (employeeSheet && employeeSheet.getLastRow() > 1) {
    // Read 3 columns now to include Phone
    const data = employeeSheet.getRange(2, 1, employeeSheet.getLastRow() - 1, 3).getValues();
    employees = data.map(row => ({
      id: row[0],
      name: row[1],
      phone: row[2] || "" // Read phone back to client
    }));
  }

  // History Log
  const historySheet = ss.getSheetByName("History Log");
  let historyLog = [];
  if (historySheet && historySheet.getLastRow() > 1) {
    const data = historySheet.getRange(2, 1, historySheet.getLastRow() - 1, 7).getValues();
    historyLog = data.map(row => ({
      id: row[0],
      timestamp: row[1],
      dateStr: row[2],
      assetId: row[3],
      assetName: row[4],
      employeeName: row[5],
      action: row[6]
    }));
  }

  const result = {
    inventory: inventory,
    employees: employees,
    history: historyLog,
    passcode: "" // You can store passcode in a "Config" sheet if you want to sync it
  };

  return ContentService.createTextOutput(JSON.stringify(result)).setMimeType(ContentService.MimeType.JSON);
}

// Helper to set up headers manually if needed
function setupSheets() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  
  let sheet = ss.getSheetByName("Inventory") || ss.insertSheet("Inventory");
  if (sheet.getLastRow() === 0) sheet.appendRow(["Asset ID", "Name", "Description", "Assigned To", "Status", "Date", "Date Assigned", "Date Returned", "Condition", "Incident Note", "Last Assigned To", "Last Date Assigned", "Last Date Returned"]);
  
  sheet = ss.getSheetByName("Employees") || ss.insertSheet("Employees");
  if (sheet.getLastRow() === 0) sheet.appendRow(["Badge ID", "Full Name", "Phone Number"]);
}
